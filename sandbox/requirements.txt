# Sandbox runtime deps — tier-1 baked scientific stack.
#
# DECOUPLED from the backend's requirements.txt / requirements.lock on purpose:
# the sandbox is its own runtime, not the app. Do NOT fold these into the app
# lockfile（backend image 的锁文件规则不适用于独立 sandbox runtime）。
#
# All pins have manylinux2014 cp311 wheels (self-contained: Pillow bundles
# libjpeg/zlib, matplotlib bundles freetype), so the image builds with no
# compiler and the intranet load needs zero network.
#
# The headline ENOSYS gamble under runsc is the C-extension set
# (numpy/pandas/matplotlib/Pillow/RapidFuzz); openpyxl/pypdf are pure-Python
# (low risk, verified anyway because xlsx/pdf are core upload scenarios).

openpyxl==3.1.5
pypdf==4.3.1
RapidFuzz==3.14.5     # text-edit / Office skill 的有界模糊定位(C-ext,与后端同版)

# 文档 skill 层：docx/pptx/pdf 的读写栈，全部使用离线 wheel。

python-docx==1.2.0    # docx 创建/修订(w:ins/w:del)/批注脚本的底座(纯 Python,lxml 之上)
